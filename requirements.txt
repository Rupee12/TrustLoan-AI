        </div>

    </div>
    """)


    # --------------------------------------------------------
    # Human-readable feature names
    # --------------------------------------------------------

    def readable_feature_name(feature):

        name = feature

        name = name.replace("num__", "")
        name = name.replace("cat__", "")

        replacements = {
            "credit_history": "Credit History",
            "applicantincome": "Applicant Income",
            "coapplicantincome": "Co-applicant Income",
            "loanamount": "Loan Amount",
            "loan_amount_term": "Loan Term",
            "dependents": "Dependents",

            "married_no": "Married — No",
            "married_yes": "Married — Yes",

            "education_graduate": "Education — Graduate",
            "education_not graduate": "Education — Not Graduate",

            "self_employed_no": "Self Employed — No",
            "self_employed_yes": "Self Employed — Yes",

            "property_area_rural": "Property Area — Rural",
            "property_area_semiurban": "Property Area — Semiurban",
            "property_area_urban": "Property Area — Urban",
        }

        if name in replacements:
            return replacements[name]

        return name.replace("_", " ").title()


    # --------------------------------------------------------
    # Prepare SHAP values
    # --------------------------------------------------------

    shap_values = explanation.values

    feature_values = explanation.data


    # Handle possible SHAP dimensions safely
    if len(np.shape(shap_values)) > 1:
        shap_values = shap_values[0]

    if len(np.shape(feature_values)) > 1:
        feature_values = feature_values[0]


    shap_df = pd.DataFrame({
        "feature": feature_names,
        "shap": shap_values,
        "value": feature_values
    })


    # Keep strongest features
    shap_df["impact"] = shap_df["shap"].abs()

    shap_df = (
        shap_df
        .sort_values("impact", ascending=False)
        .head(8)
        .sort_values("shap")
    )


    # --------------------------------------------------------
    # Create dark SHAP figure
    # --------------------------------------------------------

    fig, ax = plt.subplots(
        figsize=(12, 6)
    )


    # TrustLoan dark background
    fig.patch.set_facecolor("#080c17")
    ax.set_facecolor("#080c17")


    # --------------------------------------------------------
    # Feature names
    # --------------------------------------------------------

    labels = [
        readable_feature_name(x)
        for x in shap_df["feature"]
    ]


    values = shap_df["shap"].values


    y_positions = np.arange(len(labels))


    # --------------------------------------------------------
    # Positive / negative colors
    # --------------------------------------------------------

    positive_color = "#34d399"
    negative_color = "#60a5fa"


    bar_colors = [
        positive_color if value >= 0 else negative_color
        for value in values
    ]


    # --------------------------------------------------------
    # Bars
    # --------------------------------------------------------

    bars = ax.barh(
        y_positions,
        values,
        height=0.58,
        color=bar_colors,
        alpha=0.95
    )


    # --------------------------------------------------------
    # Zero line
    # --------------------------------------------------------

    ax.axvline(
        0,
        color="#64748b",
        linewidth=1.2,
        alpha=0.7
    )


    # --------------------------------------------------------
    # Feature labels
    # --------------------------------------------------------

    ax.set_yticks(y_positions)

    ax.set_yticklabels(
        labels,
        fontsize=11,
        color="#e2e8f0"
    )


    # --------------------------------------------------------
    # SHAP values on bars
    # --------------------------------------------------------

    max_value = max(abs(values)) if len(values) else 1

    for bar, value in zip(bars, values):

        y = bar.get_y() + bar.get_height() / 2

        if value >= 0:

            x = value + max_value * 0.025
            alignment = "left"

        else:

            x = value - max_value * 0.025
            alignment = "right"

        ax.text(
            x,
            y,
            f"{value:+.3f}",
            va="center",
            ha=alignment,
            fontsize=10,
            fontweight="bold",
            color="#f8fafc"
        )


    # --------------------------------------------------------
    # Title inside chart
    # --------------------------------------------------------

    ax.text(
        0,
        1.08,
        "Feature Contribution to Model Prediction",
        transform=ax.transAxes,
        fontsize=13,
        fontweight="bold",
        color="#f8fafc",
        ha="left"
    )


    # --------------------------------------------------------
    # Subtitle
    # --------------------------------------------------------

    ax.text(
        0,
        1.025,
        "Green = supports approval   •   Blue = reduces approval score",
        transform=ax.transAxes,
        fontsize=9,
        color="#94a3b8",
        ha="left"
    )


    # --------------------------------------------------------
    # Clean chart
    # --------------------------------------------------------

    for spine in ax.spines.values():
        spine.set_visible(False)


    ax.tick_params(
        axis="x",
        colors="#64748b",
        labelsize=9
    )

    ax.tick_params(
        axis="y",
        length=0
    )


    ax.grid(
        axis="x",
        linestyle="--",
        linewidth=0.6,
        alpha=0.12,
        color="#94a3b8"
    )


    ax.set_axisbelow(True)


    # Remove unnecessary margins
    ax.margins(y=0.08)


    plt.tight_layout()


    # --------------------------------------------------------
    # Display
    # --------------------------------------------------------

    st.pyplot(
        fig,
        width="stretch"
    )


    plt.close(fig)


    # ========================================================
    # TOP FACTORS
    # ========================================================

    st.html("""
    <div class="analysis-heading">

        <div class="analysis-number">
            02
        </div>

        <div>
            <div class="analysis-title">
                Top Decision Factors
            </div>

            <div class="analysis-subtitle">
                The strongest factors influencing the prediction
            </div>
        </div>

    </div>
    """)


    explanation_df = pd.DataFrame({

        "Feature": feature_names,

        "SHAP Value":
            explanation.values

    })


    explanation_df[
        "Absolute Impact"
    ] = explanation_df[
        "SHAP Value"
    ].abs()


    explanation_df = (
        explanation_df
        .sort_values(
            "Absolute Impact",
            ascending=False
        )
        .head(6)
    )


    factors_html = '<div class="factor-list">'

    for _, row in explanation_df.iterrows():

        feature = row["Feature"]

        value = row["SHAP Value"]

        readable_name = (
            feature
            .replace("num__", "")
            .replace("cat__", "")
            .replace("_", " ")
            .title()
        )

        if value > 0:

            icon = "↑"
            text = "Supports approval"
            factor_class = "positive"

        else:

            icon = "↓"
            text = "Reduces approval score"
            factor_class = "negative"


        factors_html += f"""
        <div class="factor-card {factor_class}">

            <div class="factor-left">

                <div class="factor-icon">
                    {icon}
                </div>

                <div>

                    <div class="factor-name">
                        {readable_name}
                    </div>

                    <div class="factor-description">
                        {text}
                    </div>

                </div>

            </div>

            <div class="factor-impact">
                {value:+.3f}
            </div>

        </div>
        """


    factors_html += "</div>"

    st.html(factors_html)
